SELECT
    r.session_id,
    r.start_time,
    DATEDIFF(MINUTE, r.start_time, GETDATE()) / 60.0 AS hours_running,
    r.status,
    r.command,
    r.database_id,
    DB_NAME(r.database_id) AS database_name,
    r.blocking_session_id,
    r.wait_type,
    r.wait_time,
    r.cpu_time,
    r.total_elapsed_time,
    r.reads,
    r.writes,
    r.logical_reads,
    s.login_name,
    s.host_name,
    s.program_name,
    t.text AS query_text
FROM sys.dm_exec_requests r
JOIN sys.dm_exec_sessions s ON r.session_id = s.session_id
CROSS APPLY sys.dm_exec_sql_text(r.sql_handle) t
WHERE DATEDIFF(MINUTE, r.start_time, GETDATE()) >= 60 * 4  -- change 4 to your X hours
ORDER BY r.start_time ASC;

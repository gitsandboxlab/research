import pandas as pd
from pathlib import Path
from datetime import date, datetime


def save_ndjson(df: pd.DataFrame, base_folder: str, run_date: date, file_name: str,
                extension: str = ".json") -> Path:
    """
    Save a DataFrame of NDJSON records (one JSON string per row, single column)
    to base_folder/YYYY/MM/DD/file_name<extension>.
    base_folder must already exist. Date folders are created if needed. Existing file is overwritten.
    """
    base = Path(base_folder)
    if not base.is_dir():
        raise FileNotFoundError(f"Base folder does not exist: {base}")

    if isinstance(run_date, datetime):
        run_date = run_date.date()

    folder = base / f"{run_date:%Y}" / f"{run_date:%m}" / f"{run_date:%d}"
    folder.mkdir(parents=True, exist_ok=True)

    if not Path(file_name).suffix:
        file_name = f"{file_name}{extension}"
    path = folder / file_name

    json_column = df.columns[0]
    with open(path, "w", encoding="utf-8", newline="\n") as f:
        for line in df[json_column]:
            f.write(f"{line}\n")

    return path

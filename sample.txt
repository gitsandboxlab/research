let targetUser = "user@contoso.com";
SigninLogs
| where TimeGenerated > ago(365d)
| where UserPrincipalName =~ targetUser
| where ResultType == "0"   // successful only
| extend LocalTime = datetime_utc_to_local(TimeGenerated, "America/Los_Angeles"),
         City      = tostring(LocationDetails.city),
         State     = tostring(LocationDetails.state),
         Country   = tostring(LocationDetails.countryOrRegion),
         Device    = strcat(tostring(DeviceDetail.displayName), " (", tostring(DeviceDetail.operatingSystem), ")")
| summarize FirstLoginPT = min(LocalTime),
            LastLoginPT  = max(LocalTime),
            Logins       = count(),
            IPs          = make_set(IPAddress, 10),
            Devices      = make_set(Device, 10)
          by DayPT = startofday(LocalTime), City, State, Country
| order by DayPT desc

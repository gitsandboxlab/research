import requests
import pandas as pd

ARG_ENDPOINT = "https://management.azure.com/providers/Microsoft.ResourceGraph/resources?api-version=2021-03-01"


def get_subassessments(assessment_key: str, subscription_id: str, credential) -> pd.DataFrame:
    """
    Fetch vulnerability sub-assessment findings for one assessment.

    Parameters
    ----------
    assessment_key : str
        The assessment's RecommendationId (the GUID from the parent
        assessment's `name` field, e.g. "1195afff-c881-495e-9bc5-1486211ae03f"
        for VM vulnerability findings) — not the full `id` resource path.
    subscription_id : str
        Subscription to scope the query to.
    credential : azure.identity.ClientSecretCredential
        Credential used to obtain a management.azure.com access token.

    Returns
    -------
    pandas.DataFrame
        One row per sub-assessment finding, one row can exist per
        (resource, CVE) pair since a single assessment_key spans every
        resource it was evaluated against. `ResourceId` (the ARM resource
        ID of the specific machine the finding belongs to) is pulled out
        as its own column; the rest of `properties` is left as a nested
        dict so the frame serializes straight to NDJSON via
        `df.to_json(orient="records", lines=True)`.
    """
    token = credential.get_token("https://management.azure.com/.default").token
    headers = {"Authorization": f"Bearer {token}", "Content-Type": "application/json"}

    query = f"""
    securityresources
    | where type =~ "microsoft.security/assessments/subassessments"
    | where id contains "/assessments/{assessment_key}/"
    | extend ResourceId = tostring(properties.resourceDetails.id)
    | project id, name, subscriptionId, resourceGroup, ResourceId, properties
    """

    records = []
    skip_token = None

    while True:
        payload = {"subscriptions": [subscription_id], "query": query}
        if skip_token:
            payload["options"] = {"$skipToken": skip_token}

        response = requests.post(ARG_ENDPOINT, headers=headers, json=payload)
        response.raise_for_status()
        body = response.json()

        records.extend(body.get("data", []))
        skip_token = body.get("$skipToken")
        if not skip_token:
            break

    return pd.DataFrame(records)

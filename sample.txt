from datetime import datetime, timedelta
from sqlalchemy import create_engine, text, event
from prefect import flow
from prefect.variables import Variable

from your_module import copy  # replace with actual import path


@flow
def hourly_materialize():
    # Full list of view->table materialization configs, stored as a Prefect variable.
    # Each record: view, table, load_mode, run_at_hour, priority, retention_months, retention_key
    records = Variable.get("view_materialization_config", default=[])
    max_records = Variable.get("max_records", default=1000000)
    now = datetime.now()

    # Figure out how long it's been since the last successful run, so we can
    # catch up on any hours we missed (e.g. if the script was down for a while).
    last_ran_raw = Variable.get("last_ran_datetime", default=None)
    last_ran = datetime.fromisoformat(last_ran_raw) if last_ran_raw else now - timedelta(hours=1)

    hours_elapsed = int((now - last_ran).total_seconds() // 3600) + 1  # always include current hour
    # run_at_hour is stored as 1-24 (hour 0 = 1, hour 23 = 24), so convert each
    # elapsed clock hour into that same 1-24 scale to build the set to match against.
    run_hours = {(now.hour - i) % 24 + 1 for i in range(hours_elapsed)}

    # Only keep records due this cycle, then sort so higher priority runs first.
    due_records = [r for r in records if r.get("run_at_hour") in run_hours]
    due_records.sort(key=lambda r: r.get("priority", 0), reverse=True)

    for record in due_records:
        # Parse '<server>.<db>.<schema>.<table>' into its 4 parts.
        view_server, view_db, view_schema, view_table = record["view"].split(".")
        dest_server, dest_db, dest_schema, dest_table = record["table"].split(".")
        load_mode = record["load_mode"]  # "append" or "truncate"

        # View and destination table can live on different servers/databases,
        # so build a separate engine for each side.
        view_engine = create_engine(f"mssql+pyodbc://{view_server}/{view_db}?driver=ODBC+Driver+17+for+SQL+Server")
        table_engine = create_engine(f"mssql+pyodbc://{dest_server}/{dest_db}?driver=ODBC+Driver+17+for+SQL+Server")

        # Set a 2-minute query timeout on the view connection just for the
        # count check below, so a slow COUNT(*) doesn't hold up the flow.
        @event.listens_for(view_engine, "connect")
        def set_count_timeout(dbapi_conn, connection_record):
            dbapi_conn.timeout = 120

        # Skip the record if the count query errors out or times out.
        try:
            with view_engine.connect() as conn:
                row_count = conn.execute(text(f"SELECT COUNT(*) FROM {view_schema}.{view_table}")).scalar()
        except Exception as e:
            print(f"Skipping {view_schema}.{view_table}: count check failed or timed out ({e})")
            continue

        # Skip the record if it has more rows than max_records allows.
        if row_count > max_records:
            print(f"Skipping {view_schema}.{view_table}: {row_count} rows exceeds max_records ({max_records})")
            continue

        # Copy the view into the destination table.
        rows_inserted = copy(
            view_engine, f"{view_schema}.{view_table}",
            table_engine, f"{dest_schema}.{dest_table}",
            load_mode,
        )
        print(f"{dest_schema}.{dest_table}: {rows_inserted} rows inserted")

        # Optional retention cleanup: delete anything older than N months
        # based on the configured date column (retention_key).
        retention_months = record.get("retention_months")
        retention_key = record.get("retention_key")
        if retention_months and retention_key:
            with table_engine.begin() as conn:
                conn.execute(text(
                    f"DELETE FROM {dest_schema}.{dest_table} "
                    f"WHERE {retention_key} < DATEADD(MONTH, -:months, GETDATE())"
                ), {"months": retention_months})

    # Mark this run's time so the next run knows how far back to catch up from.
    Variable.set("last_ran_datetime", now.isoformat(), overwrite=True)


if __name__ == "__main__":
    hourly_materialize.serve(name="hourly-view-materialize", cron="0 * * * *")

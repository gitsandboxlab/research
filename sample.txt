/* =====================================================================================
   Tableau admin views for SQL Server

   Assumes the Python output is loaded into ONE landing table:
       raw.TableauRepoRaw (RawDateTime DATETIME2, SourceTable NVARCHAR(128),
                           SourceDateTime DATETIME2, JsonRecord NVARCHAR(MAX))
   If you load one table per source instead, replace
       FROM raw.TableauRepoRaw r ... WHERE r.SourceTable = 'x'
   with
       FROM raw.<your_table_for_x> r
   in each base view.

   Requires SQL Server 2017+ (OPENJSON, STRING_AGG) and database compatibility level 130+.

   Layer 1 (tableau.base_*): one view per repository table, JSON -> typed columns
       - Snapshot tables:    rows from the latest RawDateTime for that table
       - Incremental tables: deduplicated on id, keeping the most recently loaded copy
     A key that's missing from the JSON (e.g. a column dropped in an upgrade) returns
     NULL instead of erroring, so these views won't break on schema changes.

   Layer 2 (tableau.rpt_*): joined, report-ready views

   Recommended index on the landing table:
       CREATE INDEX IX_TableauRepoRaw_Source ON raw.TableauRepoRaw (SourceTable, RawDateTime);
   ===================================================================================== */

IF SCHEMA_ID('tableau') IS NULL EXEC('CREATE SCHEMA tableau');
GO

WHERE r.SourceTable = 'sites'

CREATE OR ALTER VIEW tableau.base_historical_events AS
WITH x AS (
    SELECT r.RawDateTime, j.*,
           ROW_NUMBER() OVER (PARTITION BY j.event_id ORDER BY r.RawDateTime DESC) AS rn
    FROM raw.TableauRepoRaw r
    CROSS APPLY OPENJSON(r.JsonRecord) WITH (
        event_id              BIGINT     '$.id',
        event_type_id         INT        '$.historical_event_type_id',
        created_at            DATETIME2  '$.created_at',
        hist_actor_user_id    INT        '$.hist_actor_user_id',
        hist_target_user_id   INT        '$.hist_target_user_id',
        hist_actor_site_id    INT        '$.hist_actor_site_id',
        hist_target_site_id   INT        '$.hist_target_site_id',
        hist_project_id       INT        '$.hist_project_id',
        hist_workbook_id      INT        '$.hist_workbook_id',
        hist_view_id          INT        '$.hist_view_id',
        hist_datasource_id    INT        '$.hist_datasource_id',
        is_failure            BIT        '$.is_failure',
        duration_in_ms        INT        '$.duration_in_ms'
    ) j
    WHERE r.SourceTable = 'historical_events'
)
SELECT RawDateTime, event_id, event_type_id, created_at,
       hist_actor_user_id, hist_target_user_id, hist_actor_site_id, hist_target_site_id,
       hist_project_id, hist_workbook_id, hist_view_id, hist_datasource_id,
       is_failure, duration_in_ms
FROM x WHERE rn = 1;
GO

/* ---- Users: one row per user per site, with login, usage and group counts ---------- */
CREATE OR ALTER VIEW tableau.rpt_users AS
SELECT
    u.user_id,
    u.user_luid,
    su.system_user_id,
    su.username,
    su.friendly_name,
    su.email,
    s.site_name,
    sr.site_role_name,
    CASE WHEN su.admin_level = 10 THEN 1 ELSE 0 END          AS is_server_admin,
    CASE WHEN sr.site_role_name = 'Unlicensed' THEN 0 ELSE 1 END AS is_licensed,
    su.created_at                                            AS user_created_at,
    u.login_at                                               AS last_login_at,
    DATEDIFF(DAY, u.login_at, SYSUTCDATETIME())              AS days_since_login,
    ISNULL(vs.total_views, 0)                                AS total_views,
    ISNULL(vs.distinct_views_used, 0)                        AS distinct_views_used,
    vs.last_viewed_at,
    ISNULL(g.group_count, 0)                                 AS group_count,
    ISNULL(own.workbooks_owned, 0)                           AS workbooks_owned
FROM tableau.base_users u
JOIN tableau.base_system_users su ON su.system_user_id = u.system_user_id
JOIN tableau.base_sites s         ON s.site_id = u.site_id
LEFT JOIN tableau.base_site_roles sr ON sr.site_role_id = u.site_role_id
LEFT JOIN (
    SELECT user_id, SUM(nviews) AS total_views, COUNT(DISTINCT view_id) AS distinct_views_used,
           MAX(last_viewed_at) AS last_viewed_at
    FROM tableau.base_views_stats GROUP BY user_id
) vs ON vs.user_id = u.user_id
LEFT JOIN (
    SELECT user_id, COUNT(*) AS group_count FROM tableau.base_group_users GROUP BY user_id
) g ON g.user_id = u.user_id
LEFT JOIN (
    SELECT owner_id, COUNT(*) AS workbooks_owned FROM tableau.base_workbooks GROUP BY owner_id
) own ON own.owner_id = u.user_id;
GO

/* ---- Workbooks: inventory, ownership, usage, connections, staleness ---------------- */
CREATE OR ALTER VIEW tableau.rpt_workbooks AS
SELECT
    w.workbook_id,
    w.workbook_luid,
    w.workbook_name,
    w.repository_url,
    s.site_name,
    s.url_namespace,
    p.project_name,
    su.username                                     AS owner_username,
    su.friendly_name                                AS owner_name,
    su.email                                        AS owner_email,
    w.size_bytes / 1048576.0                        AS size_mb,
    w.data_engine_extracts                          AS has_extract,
    w.extracts_refreshed_at,
    w.created_at,
    w.updated_at,
    w.last_published_at,
    ISNULL(vc.sheet_count, 0)                       AS sheet_count,
    ISNULL(st.total_views, 0)                       AS total_views,
    ISNULL(st.distinct_viewers, 0)                  AS distinct_viewers,
    st.last_viewed_at,
    DATEDIFF(DAY, st.last_viewed_at, SYSUTCDATETIME()) AS days_since_viewed,
    CASE WHEN st.last_viewed_at IS NULL
           OR st.last_viewed_at < DATEADD(DAY, -90, SYSUTCDATETIME())
         THEN 1 ELSE 0 END                          AS is_stale_90d,
    ISNULL(sub.subscription_count, 0)               AS subscription_count,
    dc.connection_types,
    dc.connection_servers
FROM tableau.base_workbooks w
JOIN tableau.base_sites s         ON s.site_id = w.site_id
LEFT JOIN tableau.base_projects p ON p.project_id = w.project_id
LEFT JOIN tableau.base_users u    ON u.user_id = w.owner_id
LEFT JOIN tableau.base_system_users su ON su.system_user_id = u.system_user_id
LEFT JOIN (
    SELECT workbook_id, COUNT(*) AS sheet_count FROM tableau.base_views GROUP BY workbook_id
) vc ON vc.workbook_id = w.workbook_id
LEFT JOIN (
    SELECT v.workbook_id, SUM(vs.nviews) AS total_views,
           COUNT(DISTINCT vs.user_id) AS distinct_viewers, MAX(vs.last_viewed_at) AS last_viewed_at
    FROM tableau.base_views_stats vs
    JOIN tableau.base_views v ON v.view_id = vs.view_id
    GROUP BY v.workbook_id
) st ON st.workbook_id = w.workbook_id
LEFT JOIN (
    SELECT target_id AS workbook_id, COUNT(*) AS subscription_count
    FROM tableau.base_subscriptions WHERE target_type = 'Workbook' GROUP BY target_id
) sub ON sub.workbook_id = w.workbook_id
LEFT JOIN (
    SELECT owner_id,
           STRING_AGG(dbclass, ', ') WITHIN GROUP (ORDER BY dbclass) AS connection_types,
           STRING_AGG(server, ', ')  WITHIN GROUP (ORDER BY server)  AS connection_servers
    FROM (SELECT DISTINCT owner_id, dbclass, server
          FROM tableau.base_data_connections WHERE owner_type = 'Workbook') d
    GROUP BY owner_id
) dc ON dc.owner_id = w.workbook_id;
GO

/* ---- Views (sheets/dashboards): usage per view ------------------------------------- */
CREATE OR ALTER VIEW tableau.rpt_views AS
SELECT
    v.view_id,
    v.view_luid,
    v.view_name,
    v.view_title,
    v.sheettype,
    v.repository_url,
    w.workbook_id,
    w.workbook_name,
    p.project_name,
    s.site_name,
    v.created_at,
    ISNULL(st.total_views, 0)            AS total_views,
    ISNULL(st.distinct_viewers, 0)       AS distinct_viewers,
    st.last_viewed_at,
    DATEDIFF(DAY, st.last_viewed_at, SYSUTCDATETIME()) AS days_since_viewed
FROM tableau.base_views v
JOIN tableau.base_workbooks w     ON w.workbook_id = v.workbook_id
JOIN tableau.base_sites s         ON s.site_id = v.site_id
LEFT JOIN tableau.base_projects p ON p.project_id = w.project_id
LEFT JOIN (
    SELECT view_id, SUM(nviews) AS total_views, COUNT(DISTINCT user_id) AS distinct_viewers,
           MAX(last_viewed_at) AS last_viewed_at
    FROM tableau.base_views_stats GROUP BY view_id
) st ON st.view_id = v.view_id;
GO

/* ---- Per-user per-view usage detail ------------------------------------------------- */
CREATE OR ALTER VIEW tableau.rpt_view_usage_by_user AS
SELECT
    s.site_name,
    w.workbook_name,
    v.view_name,
    su.username,
    su.friendly_name,
    vs.nviews          AS view_count,
    vs.last_viewed_at,
    vs.device_type
FROM tableau.base_views_stats vs
JOIN tableau.base_views v         ON v.view_id = vs.view_id
JOIN tableau.base_workbooks w     ON w.workbook_id = v.workbook_id
JOIN tableau.base_sites s         ON s.site_id = vs.site_id
JOIN tableau.base_users u         ON u.user_id = vs.user_id
JOIN tableau.base_system_users su ON su.system_user_id = u.system_user_id;
GO

/* ---- Published data sources --------------------------------------------------------- */
CREATE OR ALTER VIEW tableau.rpt_datasources AS
SELECT
    d.datasource_id,
    d.datasource_luid,
    d.datasource_name,
    d.repository_url,
    s.site_name,
    p.project_name,
    su.username               AS owner_username,
    su.friendly_name          AS owner_name,
    d.is_certified,
    d.data_engine_extracts    AS has_extract,
    d.extracts_refreshed_at,
    d.size_bytes / 1048576.0  AS size_mb,
    d.created_at,
    d.updated_at,
    dc.connection_types,
    dc.connection_servers,
    dc.connection_databases
FROM tableau.base_datasources d
JOIN tableau.base_sites s         ON s.site_id = d.site_id
LEFT JOIN tableau.base_projects p ON p.project_id = d.project_id
LEFT JOIN tableau.base_users u    ON u.user_id = d.owner_id
LEFT JOIN tableau.base_system_users su ON su.system_user_id = u.system_user_id
LEFT JOIN (
    SELECT owner_id,
           STRING_AGG(dbclass, ', ') WITHIN GROUP (ORDER BY dbclass) AS connection_types,
           STRING_AGG(server, ', ')  WITHIN GROUP (ORDER BY server)  AS connection_servers,
           STRING_AGG(dbname, ', ')  WITHIN GROUP (ORDER BY dbname)  AS connection_databases
    FROM (SELECT DISTINCT owner_id, dbclass, server, dbname
          FROM tableau.base_data_connections WHERE owner_type = 'Datasource') x
    GROUP BY owner_id
) dc ON dc.owner_id = d.datasource_id
WHERE d.parent_workbook_id IS NULL;      -- published only (embedded sources excluded)
GO

/* ---- Historical events: logins, view access, publishes, deletes, permission changes - */
CREATE OR ALTER VIEW tableau.rpt_events AS
SELECT
    e.event_id,
    e.created_at,
    t.event_type,
    t.action_type,
    hs.site_name,
    au.username      AS actor_username,
    au.email         AS actor_email,
    tu.username      AS target_username,
    hp.project_name,
    hw.workbook_name,
    hv.view_name,
    hd.datasource_name,
    e.is_failure,
    e.duration_in_ms
FROM tableau.base_historical_events e
LEFT JOIN tableau.base_historical_event_types t ON t.event_type_id = e.event_type_id
LEFT JOIN tableau.base_hist_sites hs       ON hs.hist_site_id = e.hist_actor_site_id
LEFT JOIN tableau.base_hist_users au       ON au.hist_user_id = e.hist_actor_user_id
LEFT JOIN tableau.base_hist_users tu       ON tu.hist_user_id = e.hist_target_user_id
LEFT JOIN tableau.base_hist_projects hp    ON hp.hist_project_id = e.hist_project_id
LEFT JOIN tableau.base_hist_workbooks hw   ON hw.hist_workbook_id = e.hist_workbook_id
LEFT JOIN tableau.base_hist_views hv       ON hv.hist_view_id = e.hist_view_id
LEFT JOIN tableau.base_hist_datasources hd ON hd.hist_datasource_id = e.hist_datasource_id;
GO

/* ---- HTTP requests: page loads and load times -------------------------------------- */
CREATE OR ALTER VIEW tableau.rpt_http_requests AS
SELECT
    h.request_id,
    h.created_at,
    h.completed_at,
    DATEDIFF(MILLISECOND, h.created_at, h.completed_at) AS duration_ms,
    h.controller,
    h.action,
    h.currentsheet,
    h.http_status,
    s.site_name,
    su.username,
    h.vizql_session,
    h.user_ip
FROM tableau.base_http_requests h
LEFT JOIN tableau.base_sites s         ON s.site_id = h.site_id
LEFT JOIN tableau.base_users u         ON u.user_id = h.user_id
LEFT JOIN tableau.base_system_users su ON su.system_user_id = u.system_user_id;
GO

/* ---- Background jobs: extract refreshes, subscriptions, status and duration -------- */
CREATE OR ALTER VIEW tableau.rpt_background_jobs AS
SELECT
    b.job_id,
    b.job_name,
    b.title           AS object_name,
    b.subtitle        AS object_type,
    s.site_name,
    CASE b.finish_code
        WHEN 0 THEN 'Success'
        WHEN 1 THEN 'Failed'
        WHEN 2 THEN 'Cancelled'
        ELSE CASE WHEN b.completed_at IS NULL THEN 'Running/Queued' ELSE 'Unknown' END
    END               AS job_status,
    b.created_at      AS queued_at,
    b.started_at,
    b.completed_at,
    DATEDIFF(SECOND, b.created_at, b.started_at)   AS queue_wait_seconds,
    DATEDIFF(SECOND, b.started_at, b.completed_at) AS run_seconds,
    b.priority,
    b.notes           AS error_notes
FROM tableau.base_background_jobs b
LEFT JOIN tableau.base_sites s ON s.site_id = b.site_id;
GO

import json
import requests
import pandas as pd
from datetime import datetime, timedelta

LP_URL = "https://lastpass.com/enterpriseapi.php"


def _lp_post(cid: int, provhash: str, cmd: str, data: dict | None = None) -> dict:
    payload = {"cid": cid, "provhash": provhash, "cmd": cmd}
    if data is not None:
        payload["data"] = data

    r = requests.post(LP_URL, json=payload, timeout=300)
    r.raise_for_status()

    if not r.text.strip():
        raise RuntimeError(f"LastPass {cmd}: empty response (check cid/provhash)")
    try:
        body = r.json()
    except ValueError:
        raise RuntimeError(f"LastPass {cmd}: non-JSON response: {r.text[:300]}")

    if isinstance(body, dict) and body.get("status") == "FAIL":
        raise RuntimeError(f"LastPass {cmd} error: {body}")
    return body


def fetch_lastpass_logs(cid: int, provhash: str, start: datetime, end: datetime,
                        window_hours: int = 6, user: str = "", search: str = "") -> pd.DataFrame:
    """Reporting events between start and end. Times are in the LastPass account timezone."""
    rows = []
    t = start
    while t < end:
        t_end = min(t + timedelta(hours=window_hours), end)
        data = {
            "from": t.strftime("%Y-%m-%d %H:%M:%S"),
            "to": t_end.strftime("%Y-%m-%d %H:%M:%S"),
            "search": search,
            "user": user,
            "format": "siem",
        }
        while True:
            body = _lp_post(cid, provhash, "reporting", data)
            events = body.get("events") or body.get("data") or []
            if isinstance(events, dict):
                events = list(events.values())
            rows.extend(events)

            nxt = body.get("next")
            if not nxt:
                break
            data["next"] = nxt
        t = t_end

    df = pd.DataFrame(rows)
    if not df.empty:
        df = df.drop(columns=["id"], errors="ignore")
        df["Time"] = pd.to_datetime(df["Time"], errors="coerce")
        df = df.drop_duplicates().sort_values("Time").reset_index(drop=True)
    return df


def fetch_lastpass_users(cid: int, provhash: str) -> pd.DataFrame:
    """One row per user."""
    body = _lp_post(cid, provhash, "getuserdata", {})
    user_data = body.get("Users") or {}

    rows = []
    for uid, u in user_data.items():
        row = {"uid": uid}
        row.update(u)
        rows.append(row)

    df = pd.DataFrame(rows)
    if not df.empty and "groups" in df.columns:
        df["groups"] = df["groups"].apply(json.dumps)  # land as JSON string
    return df


def fetch_lastpass_groups(cid: int, provhash: str) -> pd.DataFrame:
    """One row per group/member. Empty groups kept with username = None."""
    body = _lp_post(cid, provhash, "getuserdata", {})
    group_data = body.get("Groups") or {}

    rows = []
    for groupname, members in group_data.items():
        if not members:
            rows.append({"groupname": groupname, "username": None})
            continue
        for username in members:
            rows.append({"groupname": groupname, "username": username})

    return pd.DataFrame(rows, columns=["groupname", "username"])


def fetch_lastpass_folders(cid: int, provhash: str) -> pd.DataFrame:
    """One row per shared folder."""
    body = _lp_post(cid, provhash, "getsfdata")

    rows = []
    for shareid, f in body.items():
        if not isinstance(f, dict):
            continue
        rows.append({
            "shareid": shareid,
            "sharedfoldername": f.get("sharedfoldername"),
            "score": f.get("score"),
            "deleted": f.get("deleted"),
        })
    return pd.DataFrame(rows)


def fetch_lastpass_folder_users(cid: int, provhash: str) -> pd.DataFrame:
    """One row per shared folder/user with permissions."""
    body = _lp_post(cid, provhash, "getsfdata")

    rows = []
    for shareid, f in body.items():
        if not isinstance(f, dict):
            continue
        for u in f.get("users") or []:
            row = {"shareid": shareid}
            row.update(u)
            rows.append(row)
    return pd.DataFrame(rows)

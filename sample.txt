import requests
import pandas as pd
from datetime import datetime, timedelta

LP_URL = "https://lastpass.com/enterpriseapi.php"


def fetch_lastpass_logs(cid: int, provhash: str, start: datetime, end: datetime,
                        window_hours: int = 6, user: str = "", search: str = "") -> pd.DataFrame:
    """Pull LastPass reporting events between start and end. Times are in the admin account's timezone."""
    rows = []
    t = start
    while t < end:
        t_end = min(t + timedelta(hours=window_hours), end)
        payload = {
            "cid": cid,
            "provhash": provhash,
            "cmd": "reporting",
            "data": {
                "from": t.strftime("%Y-%m-%d %H:%M:%S"),
                "to": t_end.strftime("%Y-%m-%d %H:%M:%S"),
                "search": search,
                "user": user,
                "format": "siem",
            },
        }
        while True:
            r = requests.post(LP_URL, json=payload, timeout=120)
            r.raise_for_status()
            body = r.json()
            if body.get("status") != "OK":
                raise RuntimeError(f"LastPass error: {body}")
            data = body.get("data") or {}
            rows.extend(data.values() if isinstance(data, dict) else data)
            nxt = body.get("next")
            if not nxt:
                break
            payload["data"]["next"] = nxt
        t = t_end

    df = pd.DataFrame(rows)
    if not df.empty and "Time" in df.columns:
        df["Time"] = pd.to_datetime(df["Time"], errors="coerce")
        df = df.drop_duplicates().sort_values("Time").reset_index(drop=True)
    return df

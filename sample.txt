try:
        content = si.RetrieveContent()
        view = content.viewManager.CreateContainerView(
            content.rootFolder, [vim.VirtualMachine], True)

        rows = []
        for vm in view.view:
            if not vm.snapshot:
                continue
            size = snapshot_size_gb(vm)
            for row in walk_snapshots(vm.snapshot.rootSnapshotList, vm):
                row["vm_snapshot_total_gb"] = size
                rows.append(row)
        view.Destroy()

        rows.sort(key=lambda r: r["created"])
        for r in rows:
            print(f"{r['vm']:<30} {r['snapshot']:<35} {r['created']:%Y-%m-%d %H:%M}  "
                  f"{r['age_days']:>4}d  {r['vm_snapshot_total_gb']:>7} GB")

        if rows:
            with open("snapshots.csv", "w", newline="") as fh:
                writer = csv.DictWriter(fh, fieldnames=rows[0].keys())
                writer.writeheader()
                writer.writerows(rows)
    finally:
        Disconnect(si)

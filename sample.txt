import json
import requests
import pandas as pd
from datetime import datetime, timedelta

LP_URL = "https://lastpass.com/enterpriseapi.php"


def _lp_post(cid: int, provhash: str, cmd: str, data: dict | None = None) -> dict:
    payload = {"cid": cid, "provhash": provhash, "cmd": cmd}
    if data is not None:
        payload["data"] = data
    r = requests.post(LP_URL, json=payload, timeout=300)
    r.raise_for_status()
    body = r.json()
    if isinstance(body, dict) and body.get("status") == "FAIL":
        raise RuntimeError(f"LastPass {cmd} error: {body}")
    return body


def fetch_lastpass_logs(cid: int, provhash: str, start: datetime, end: datetime,
                        window_hours: int = 6, user: str = "", search: str = "") -> pd.DataFrame:
    """Reporting events between start and end. Times are in the LastPass account timezone."""
    rows = []
    t = start
    while t < end:
        t_end = min(t + timedelta(hours=window_hours), end)
        data = {
            "from": t.strftime("%Y-%m-%d %H:%M:%S"),
            "to": t_end.strftime("%Y-%m-%d %H:%M:%S"),
            "search": search,
            "user": user,
            "format": "siem",
        }
        while True:
            body = _lp_post(cid, provhash, "reporting", data)
            events = body.get("data") or {}
            rows.extend(events.values() if isinstance(events, dict) else events)
            nxt = body.get("next")
            if not nxt:
                break
            data["next"] = nxt
        t = t_end

    df = pd.DataFrame(rows)
    if not df.empty:
        df = df.drop(columns=["id"], errors="ignore")
        df["Time"] = pd.to_datetime(df["Time"], errors="coerce")
        df = df.drop_duplicates().sort_values("Time").reset_index(drop=True)
    return df


def fetch_lastpass_users(cid: int, provhash: str) -> tuple[pd.DataFrame, pd.DataFrame]:
    """Returns (users, groups). getuserdata with no username returns all users plus group membership."""
    body = _lp_post(cid, provhash, "getuserdata", {})

    users = pd.DataFrame([{"uid": uid, **u} for uid, u in (body.get("Users") or {}).items()])
    if not users.empty and "groups" in users.columns:
        users["groups"] = users["groups"].apply(json.dumps)  # land as JSON string

    groups = pd.DataFrame(
        [{"groupname": g, "username": u}
         for g, members in (body.get("Groups") or {}).items()
         for u in (members or [None])],  # keep empty groups
        columns=["groupname", "username"],
    )
    return users, groups


def fetch_lastpass_folders(cid: int, provhash: str) -> tuple[pd.DataFrame, pd.DataFrame]:
    """Returns (folders, folder_users) from getsfdata."""
    body = _lp_post(cid, provhash, "getsfdata")

    folders, folder_users = [], []
    for sfid, f in body.items():
        if not isinstance(f, dict):
            continue
        folders.append({"shareid": sfid, "sharedfoldername": f.get("sharedfoldername"),
                        "score": f.get("score"), "deleted": f.get("deleted")})
        for u in f.get("users") or []:
            folder_users.append({"shareid": sfid, **u})

    return pd.DataFrame(folders), pd.DataFrame(folder_users)
